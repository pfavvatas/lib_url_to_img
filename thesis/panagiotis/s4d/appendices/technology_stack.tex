\section{Technology Stack Overview}
\label{sec:tech-overview}

The S\textsuperscript{4}D system leverages a comprehensive technology stack designed for scalability, performance, and maintainability. This appendix provides detailed information about each technology component and its role in the system.

\section{Backend Technologies}
\label{sec:backend-tech}

\subsection{Core Programming Language}
\label{subsec:core-language}

\textbf{Python 3.8+}
\begin{itemize}
    \item \textbf{Purpose}: Primary backend development language
    \item \textbf{Advantages}: 
        \begin{itemize}
            \item Rich ecosystem for data science and web development
            \item Excellent libraries for machine learning and clustering
            \item Strong web automation capabilities
            \item Readable and maintainable code structure
        \end{itemize}
    \item \textbf{Usage in S4D}: All core processing, clustering algorithms, API development
\end{itemize}

\subsection{Web Automation Framework}
\label{subsec:selenium}

\textbf{Selenium WebDriver 4.x}
\begin{itemize}
    \item \textbf{Purpose}: Web page automation and DOM extraction
    \item \textbf{Configuration}: 
        \begin{itemize}
            \item Chrome browser in headless mode
            \item Window size: 1920x1080 for consistent rendering
            \item No sandbox mode for Linux compatibility
            \item GPU disabled for server environments
        \end{itemize}
    \item \textbf{Features Used}:
        \begin{itemize}
            \item DOM tree traversal
            \item CSS property extraction
            \item JavaScript execution for computed styles
            \item Element bounding box calculations
        \end{itemize}
    \item \textbf{Usage in S4D}: URL processing, HTML structure extraction, CSS analysis
\end{itemize}

\subsection{API Framework}
\label{subsec:api-framework}

\textbf{FastAPI 0.100+}
\begin{itemize}
    \item \textbf{Purpose}: REST API development and documentation
    \item \textbf{Key Features}:
        \begin{itemize}
            \item Automatic OpenAPI/Swagger documentation
            \item Type hints and validation
            \item Async/await support
            \item High performance (comparable to NodeJS)
        \end{itemize}
    \item \textbf{Configuration}:
        \begin{itemize}
            \item CORS enabled for React frontend
            \item JSON request/response handling
            \item Error handling middleware
            \item Request timeout management
        \end{itemize}
    \item \textbf{Usage in S4D}: All API endpoints, experiment management, file serving
\end{itemize}

\subsection{Data Processing Libraries}
\label{subsec:data-processing}

\textbf{NumPy 1.24+}
\begin{itemize}
    \item \textbf{Purpose}: Numerical computing and array operations
    \item \textbf{Usage}: Vector operations, similarity matrix calculations, data normalization
\end{itemize}

\textbf{Pandas 2.0+}
\begin{itemize}
    \item \textbf{Purpose}: Data manipulation and analysis
    \item \textbf{Usage}: Role vector computation, similarity data organization, result aggregation
\end{itemize}

\textbf{SciPy 1.10+}
\begin{itemize}
    \item \textbf{Purpose}: Scientific computing algorithms
    \item \textbf{Usage}: Distance calculations, statistical functions, sparse matrix operations
\end{itemize}

\subsection{Machine Learning Libraries}
\label{subsec:ml-libraries}

\textbf{Scikit-learn 1.3+}
\begin{itemize}
    \item \textbf{Purpose}: Machine learning algorithms and preprocessing
    \item \textbf{Components Used}:
        \begin{itemize}
            \item StandardScaler for data normalization
            \item Cosine similarity calculations
            \item PCA for dimensionality reduction
            \item Various distance metrics
        \end{itemize}
    \item \textbf{Usage in S4D}: Data preprocessing, similarity computation, feature scaling
\end{itemize}

\textbf{HDBSCAN 0.8+}
\begin{itemize}
    \item \textbf{Purpose}: Density-based clustering algorithm
    \item \textbf{Key Features}:
        \begin{itemize}
            \item Hierarchical clustering
            \item Automatic cluster count determination
            \item Outlier detection and handling
            \item DBCV quality scoring
        \end{itemize}
    \item \textbf{Configuration}:
        \begin{itemize}
            \item Dynamic min\_cluster\_size optimization
            \item Cluster selection epsilon tuning
            \item Outlier reassignment to nearest clusters
        \end{itemize}
    \item \textbf{Usage in S4D}: Primary clustering algorithm, quality assessment
\end{itemize}

\textbf{DBCV (Density-Based Cluster Validation)}
\begin{itemize}
    \item \textbf{Purpose}: Clustering quality evaluation
    \item \textbf{Features}: Density-aware cluster validation metric
    \item \textbf{Usage in S4D}: Clustering quality scoring, parameter optimization
\end{itemize}

\subsection{Caching System}
\label{subsec:caching-system}

\textbf{File-based Caching}
\begin{itemize}
    \item \textbf{Purpose}: Performance optimization through result reuse
    \item \textbf{Implementation}:
        \begin{itemize}
            \item SHA-256 hash-based cache keys
            \item TTL-based expiration (24h CLI, 1h API)
            \item Partial and complete cache support
            \item Cache statistics and monitoring
        \end{itemize}
    \item \textbf{Benefits}: 50-90\% processing time reduction for cached URLs
\end{itemize}

\section{System Requirements}
\label{sec:system-requirements}

\subsection{Minimum Requirements}
\label{subsec:minimum-requirements}

\textbf{Backend Server}:
\begin{itemize}
    \item \textbf{OS}: Linux, macOS, or Windows
    \item \textbf{Python}: 3.8 or higher
    \item \textbf{Memory}: 4GB RAM minimum, 8GB recommended
    \item \textbf{Storage}: 2GB free space for cache and results
    \item \textbf{Network}: Stable internet connection
\end{itemize}

\textbf{Frontend Client}:
\begin{itemize}
    \item \textbf{Browser}: Modern browser with JavaScript support
    \item \textbf{Resolution}: 1024x768 minimum
    \item \textbf{Network}: Stable connection to backend API
    \item \textbf{Memory}: 2GB RAM for complex visualizations
\end{itemize}

\subsection{Recommended Configuration}
\label{subsec:recommended-configuration}

\textbf{Production Server}:
\begin{itemize}
    \item \textbf{CPU}: Multi-core processor (4+ cores)
    \item \textbf{Memory}: 16GB+ RAM for large-scale processing
    \item \textbf{Storage}: SSD for improved I/O performance
    \item \textbf{Network}: High-bandwidth connection
    \item \textbf{Browser}: Chrome installed for Selenium automation
\end{itemize}
