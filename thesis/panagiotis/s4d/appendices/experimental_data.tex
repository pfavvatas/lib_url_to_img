\section{Statistical Significance Tests}
\label{sec:statistical-tests}

\subsection{Hypothesis Testing Results}
\label{subsec:hypothesis-testing}

\textbf{H1}: S4D clustering is significantly better than random clustering
\begin{itemize}
    \item \textbf{Test}: Paired t-test with p < 0.001
    \item \textbf{Effect size}: 2.34 (large effect)
    \item \textbf{Conclusion}: Highly significant improvement over baseline
\end{itemize}

\textbf{H2}: Multi-level analysis improves clustering quality
\begin{itemize}
    \item \textbf{Test}: One-way ANOVA with F = 15.67, p < 0.001
    \item \textbf{Post-hoc}: Level 5 significantly superior to shallow levels
    \item \textbf{Conclusion}: Depth of analysis is crucial for quality
\end{itemize}

\textbf{H3}: Cache system significantly improves performance
\begin{itemize}
    \item \textbf{Test}: Wilcoxon signed-rank test
    \item \textbf{W-statistic}: 1,247
    \item \textbf{p-value}: < 0.001
    \item \textbf{Conclusion}: Highly significant performance improvement
\end{itemize}
